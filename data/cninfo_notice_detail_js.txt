				if (!value || value == '-') {
					return '-'
				}
				return Number(value).toFixed(2) + "%";

				var title = this.announcementTitle;

				        // 使用 XMLHttpRequest 实现进度监控
				        const xhr = new XMLHttpRequest();
				        xhr.open('GET', this.pdfUrl, true);
				        xhr.responseType = 'blob';

				        // 进度监控
				        xhr.addEventListener('progress', function(event) {
							const percent = Math.floor((event.loaded / event.total)*100);
							if(percent > 5){
								_this.percent = percent
							}
				        	if(event.total == event.loaded){
								setTimeout(function()  {
									_this.isDownloading = false;
									_this.percent = 5
								}, 1000);
							}
				        });

				        // 完成处理
				        xhr.addEventListener('load',function (event){
				          if (xhr.status === 200) {
				            // 创建 Blob 对象
				            var blob = new Blob([xhr.response]);

				            // 创建临时链接
				            var link = document.createElement('a');
				            link.href = window.URL.createObjectURL(blob);
				            link.download = title+".pdf"; // 设置下载文件名

				            // 触发下载
				            document.body.appendChild(link);
				            var clickEvent = document.createEvent('MouseEvents');
                            clickEvent.initEvent('click', true, true);
                            link.dispatchEvent(clickEvent);
				            document.body.removeChild(link);
				          }
				        });

				        // 错误处理
				        xhr.addEventListener('error', function() {
				          this.$message.error('下载失败');
				          _this.isDownloading = false;
				          _this.percent = 5
				        });

				        xhr.send();
				      } catch (error) {

						_this.percent = 5

						this.announcementTime = res.announcement.announcementTime;

							B: '早间公告',

					height: "100%",

					width: 70,

			},
			getViewNum: function() {
				axios({
					url: path + '/announcement/getCount',
					method: 'get',
					params: {
						announceId: announcementId
					}
				}).then(function(res) {
					if (res) {
						this.viewNum = res.clickTotalTimes
					}
				}.bind(this)).catch(function(err) {
					this.viewNum = '';
				}.bind(this))
			},
			getStockPrice: function() {
				JSONP.getJSON(apiServer + '/v5/hq/dataItem', {
					jsonpCallback: '',
					codelist: { szse: 'sz', sse: 'sh' }[plate] + stockCode

				}, function(res) {
					if (res && res[0]) {
						var val = res[0]
						this.price = val[10] ? val[10] : '-';
						this.isRise = val[264648] > 0 ? true : false;
						this.change = (val[264648] ? parseFloat(val[264648]).toFixed(2) : '-') + (val[199112] ? ' (' + parseFloat(val[199112]).toFixed(2) + '%)' : '-')

				}.bind(this))
			},
			query: function() {

				$.ajax({
					url: path + '/singleDisclosure/getStockPlateNew',
					type: 'get',
					data: stocklist,
					success: function(data) {
						if (data.msg != "fail") {
							var prefix = data.content.market;
							if (prefix.indexOf("bj") != -1) {

							} else {
								var code = stockCode + '.' + (prefix === "sz" ? "SZ" : "SH");

									// 获取行情订阅数据

								});
							}
						}
					}
				});

			// 获取行情订阅数据
			getSubscribeData: function(code) {
				this.subscribe = NSDK.createRequestItem(SDK_REQUEST_QUOTESUBSCRIBE);
				this.subscribe.setDataCallback(this.setSubscribeData);

				this.subscribe.setFields(
					"Last,Change,PercentChange"
				);
				this.subscribe.request();

				_this.change = (rowArray[2] ? rowArray[2] : '-') + (rowArray[3] && rowArray[3] != '-' ? '(' + _this.toPercent(rowArray[3]) + ')' : '-');
				NSDK.disconnect();

						}.bind(this)

					return;

				params.append('content', '##' + this.askPlaceHolder + '##' + this.askMsg);

				params.append('mobile', JC_USER.getInfo('mobile'));
				params.append('email', JC_USER.getInfo('email'));
				axios({

					data: params

							title: '提示',

				}.bind(this))

				}.bind(this))

								}
							}
						})
					}
				}
			}
			if (plate == 'fund') {
				this.getCodeListByAid(announcementId)
			}
		},
		mounted: function() {
			var _this = this
			this.getNoticeInfo();

		beforeDestroy: function() {
			var _this = this
			if (hq_api_enable !== 'true' && myBrowser() != 'IE') {
				NSDK.destroyRequestItem(_this.subscribe);
			}
		},
